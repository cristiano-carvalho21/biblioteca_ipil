def perfis_podem_emprestar():
    perfis = Perfil.objects.all()

    permitidos = []
    nao_permitidos = []

    for p in perfis:
        # tentativa segura (não quebra se não existir campo)
        pode = getattr(p, "pode_emprestar", None)

        if pode is True:
            permitidos.append(p.nome if hasattr(p, "nome") else str(p))
        elif pode is False:
            nao_permitidos.append(p.nome if hasattr(p, "nome") else str(p))

    if permitidos:
        return "✅ Perfis que podem emprestar:\n" + "\n".join(f"• {p}" for p in permitidos)

    return "ℹ️ Não há regra explícita de perfis que podem emprestar."


def perfis_nao_podem_emprestar():
    perfis = Perfil.objects.all()

    bloqueados = []

    for p in perfis:
        pode = getattr(p, "pode_emprestar", None)

        if pode is False:
            bloqueados.append(p.nome if hasattr(p, "nome") else str(p))

    if bloqueados:
        return "⛔ Perfis que NÃO podem emprestar:\n" + "\n".join(f"• {p}" for p in bloqueados)

    return "ℹ️ Não há perfis explicitamente bloqueados para empréstimos."


def info_multas():
    config = ConfiguracaoSistema.objects.first()

    if not config:
        return "ℹ️ Nenhuma configuração de multas definida."

    multa_diaria = getattr(config, "multa_diaria", None)
    dias_graca = getattr(config, "dias_graca", None)

    partes = ["📌 Regras de multas:"]

    if multa_diaria is not None:
        partes.append(f"• Multa diária: {multa_diaria}")

    if dias_graca is not None:
        partes.append(f"• Dias de tolerância: {dias_graca}")

    if len(partes) == 1:
        return "ℹ️ Sistema não tem regras de multas configuradas."

    return "\n".join(partes)


    # =========================
# 🏛️ REGRAS ADMINISTRATIVAS
# =========================

if any(p in texto for p in ["quem pode emprestar", "pode emprestar", "empréstimo quem", "emprestar livros"]):
    return [perfis_podem_emprestar()]

if any(p in texto for p in ["quem não pode emprestar", "não pode emprestar", "bloqueado empréstimo"]):
    return [perfis_nao_podem_emprestar()]

if any(p in texto for p in ["multas", "multa", "penalidade", "atraso", "juros"]):
    return [info_multas()]
