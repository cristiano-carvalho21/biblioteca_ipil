def dividir_perguntas(texto):
    separadores = r'[?.,;]| e | também | alem disso | depois |\n'
    partes = re.split(separadores, texto)

    perguntas = [
        p.strip() for p in partes
        if p.strip() and len(p.strip()) > 2
    ]

    return perguntas

def limpar_texto(texto):
    texto = normalizar(texto)
    return reduzir_repeticoes(texto).strip()


def formatar_lista(qs, campo="titulo", vazio="Sem resultados."):
    itens = list(qs)
    if not itens:
        return vazio

    return "\n".join(
        f"• {getattr(i, campo)}"
        for i in itens if getattr(i, campo, None)
    )


def dividir_perguntas(texto):
    return [
        p.strip()
        for p in re.split(r"[?.,;]| e | também | alem disso | depois |\n", texto)
        if len(p.strip()) > 1
    ]

def extrair_categoria(texto):
    categorias = list(Categoria.objects.values_list("nome", flat=True))
    from rapidfuzz import process

    if not categorias:
        return None

    m = process.extractOne(texto, categorias)
    return m[0] if m and m[1] > 70 else None


def extrair_autor(texto):
    autores = list(Autor.objects.values_list("nome", flat=True))
    from rapidfuzz import process

    if not autores:
        return None

    m = process.extractOne(texto, autores)
    return m[0] if m and m[1] > 70 else None

def e_intencao_recomendacao(texto):
    return any(p in texto for p in [
        "recomenda", "recomendar", "sugere", "quero ler",
        "livros bons", "indica"
    ])


def recomendar_livros(texto):

    categoria = extrair_categoria(texto)

    # 1 categoria
    if categoria:
        livros = Livro.objects.filter(categoria__nome__icontains=categoria)[:6]
        if livros:
            return f"📚 Recomendo de {categoria}:\n{formatar_lista(livros)}"

    # 2 conteúdo
    q = Q()
    for p in texto.split():
        q |= Q(titulo__icontains=p)
        q |= Q(descricao__icontains=p)
        q |= Q(sumario__icontains=p)

    livros = Livro.objects.filter(q).distinct()[:6]

    if livros:
        return f"📚 Recomendações:\n{formatar_lista(livros)}"

    # 3 popular
    livros = Livro.objects.annotate(
        total=Count("reservas")
    ).order_by("-total")[:6]

    if livros:
        return f"📚 Populares:\n{formatar_lista(livros)}"

    return "📚 Sem recomendações suficientes."

def consultas_confidenciais(user, texto):

    ok, msg = proteger_dados(user)
    if not ok:
        return [msg]

    if "alunos" in texto:
        from accounts.models import AlunoOficial
        return [f"👨‍🎓 Alunos: {AlunoOficial.objects.count()}"]

    if "funcionarios" in texto:
        from accounts.models import FuncionarioOficial
        return [f"👨‍🏫 Funcionários: {FuncionarioOficial.objects.count()}"]

    if "emprestimos em atraso" in texto:
        total = Emprestimo.objects.filter(acoes="atrasado").count()
        return [f"⏰ Atrasados: {total}"]

    if "emprestimos ativos" in texto:
        total = Emprestimo.objects.filter(acoes="ativo").count()
        return [f"📦 Ativos: {total}"]

    if "livros em uso" in texto:
        total = Emprestimo.objects.filter(acoes="ativo").count()
        return [f"📚 Em uso: {total}"]

    return None

def processar_pergunta_unica(texto, user, chat):

    memoria = obter_memoria_inteligente(chat)
    texto = limpar_texto(texto)

    # 🔐 PRIMEIRO: CONFIDENCIAL
    conf = consultas_confidenciais(user, texto)
    if conf:
        return conf

    # 📚 RECOMENDAÇÃO
    if e_intencao_recomendacao(texto):
        return [recomendar_livros(texto)]

    # 📊 RELATÓRIO
    if "relatorio" in texto:
        return [gerar_relatorio(user)]

    # =========================
    # 🔥 TEU SISTEMA ORIGINAL (INTACTO)
    # =========================

    if "livros" in texto:
        return [f"📚 {livros_lista()}"]

    if "categorias" in texto:
        return [f"📂 {categorias_lista()}"]

    if "autores" in texto:
        return [f"👤 {autores_lista()}"]

    if "exposicoes" in texto:
        return [f"🖼️ {exposicoes_lista()}"]

    if "eventos" in texto:
        return [f"🎉 {eventos_lista()}"]

    # RAG
    resultados = buscar_livros(texto)

    if resultados:
        return ["📚 Encontrado:\n" +
                "\n".join(f"• {r['livro'].titulo}" for r in resultados[:5])]

    return ["🤔 Não entendi."]
